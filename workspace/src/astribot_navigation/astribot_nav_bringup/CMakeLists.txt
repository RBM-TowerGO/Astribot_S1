cmake_minimum_required(VERSION 3.8)
project(astribot_nav_bringup)
find_package(ament_cmake REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_library(towergo_exact_goal_lattice SHARED src/exact_goal_lattice.cpp)
target_compile_features(towergo_exact_goal_lattice PUBLIC cxx_std_17)
ament_target_dependencies(towergo_exact_goal_lattice nav2_smac_planner pluginlib tf2_geometry_msgs)
target_link_libraries(towergo_exact_goal_lattice Eigen3::Eigen)
pluginlib_export_plugin_description_file(nav2_core planner_plugins.xml)
install(TARGETS towergo_exact_goal_lattice LIBRARY DESTINATION lib)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES ../scripts/nav_web.py ../scripts/nav_web.html DESTINATION share/${PROJECT_NAME}/web)
find_package(nav2_amcl REQUIRED)
add_library(towergo_lio_motion_model SHARED src/lio_planar_motion_model.cpp)
target_compile_features(towergo_lio_motion_model PUBLIC cxx_std_17)
ament_target_dependencies(towergo_lio_motion_model nav2_amcl pluginlib)
pluginlib_export_plugin_description_file(nav2_amcl motion_plugins.xml)
install(TARGETS towergo_lio_motion_model LIBRARY DESTINATION lib)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(std_msgs REQUIRED)
add_library(towergo_stop_turn_planner SHARED src/stop_turn_planner.cpp)
add_library(towergo_stop_turn_controller SHARED src/stop_turn_controller.cpp)
foreach(target towergo_stop_turn_planner towergo_stop_turn_controller)
  target_compile_features(${target} PUBLIC cxx_std_17)
  ament_target_dependencies(${target} nav2_core nav2_costmap_2d pluginlib tf2_geometry_msgs std_msgs)
endforeach()
pluginlib_export_plugin_description_file(nav2_core stop_turn_plugins.xml)
install(TARGETS towergo_stop_turn_planner towergo_stop_turn_controller LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(stop_turn_geometry_test src/stop_turn_geometry_test.cpp)
  target_compile_features(stop_turn_geometry_test PUBLIC cxx_std_17)
  ament_target_dependencies(stop_turn_geometry_test nav2_costmap_2d)
  add_test(NAME stop_turn_geometry COMMAND stop_turn_geometry_test)
endif()
ament_package()
