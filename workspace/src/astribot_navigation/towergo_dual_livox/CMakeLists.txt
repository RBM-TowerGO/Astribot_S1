cmake_minimum_required(VERSION 3.8)
project(towergo_dual_livox)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(dual_adapter src/dual_adapter.cpp)
target_include_directories(dual_adapter PRIVATE include)
ament_target_dependencies(dual_adapter rclcpp sensor_msgs std_msgs diagnostic_msgs livox_ros_driver2)
target_compile_options(dual_adapter PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS dual_adapter DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_fusion tests/test_fusion.cpp)
  target_include_directories(test_fusion PRIVATE include)
  target_compile_options(test_fusion PRIVATE -UNDEBUG)
  add_test(NAME fusion COMMAND test_fusion)
endif()
ament_package()
