cmake_minimum_required(VERSION 3.8)
project(towergo_livox_adapter)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(livox_adapter src/livox_adapter.cpp)
ament_target_dependencies(livox_adapter rclcpp sensor_msgs std_msgs diagnostic_msgs livox_ros_driver2)
target_compile_options(livox_adapter PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS livox_adapter DESTINATION lib/${PROJECT_NAME})
ament_package()
